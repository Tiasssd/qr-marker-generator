    60→/** Задание на печать: одна метка или лист со всеми метками из истории */
    61→interface PrintJob {
    62→  /** data-URL картинок меток */
    63→  images: string[];
    64→  /** Понятные подписи (названия) — для alt */
    65→  labels: string[];
    66→  /** Физическая ширина одной метки на бумаге, см */

    72→function signature(it: MarkerConfig): string {
    73→  // ID и есть идентичность метки: «Молния» и «молния» — один и тот же узор
    74→  return it.id;
    75→}
    76→
    77→/* История: v2 (ID выводится из названия) с миграцией из v1 */
    78→function loadHistory(): HistoryItem[] {
    79→  try {
    80→    const raw =
    81→      window.localStorage.getItem(HISTORY_KEY) ??
    82→      window.localStorage.getItem("ar-marker-studio:history:v1");
    83→    if (!raw) return [];
    84→    const parsed: unknown = JSON.parse(raw);
    85→    if (!Array.isArray(parsed)) return [];
    86→    const items = parsed
    87→      .filter(
    88→        (it): it is HistoryItem =>
    89→          Boolean(it) &&
    90→          typeof it === "object" &&
    91→          typeof (it as HistoryItem).name === "string"
    92→      )
    93→      // Название — теперь идентичность метки: ID пересчитываем из него
    94→      .map((it) => {
    95→        const name = it.name.trim().replace(/\s+/g, " ");
    96→        return { name, id: deriveId(name), ts: typeof it.ts === "number" ? it.ts : Date.now() };
    97→      })
    98→      .filter((it) => it.name.length > 0);
    99→    return items.slice(0, 40);
   100→  } catch {
   101→    return [];
   102→  }
   103→}

   176→      dpr: Math.min(window.devicePixelRatio || 1, 2),

   302→

   313→      saveBlob(blob, markerFileName(item.name, item.id));
   314→      toast({ title: "PNG сохранён", description: `Метка «${item.name}».`, variant: "success", duration: 2500 });
   315→    } catch {
   316→      toast({ title: "Не удалось создать PNG", variant: "destructive", duration: 3500 });
   317→    }
   318→  };
   319→

   323→      title: "Метка удалена из истории",
   324→      description: `«${item.name}»`,

   333→      ),
   334→    });
   335→  };
   336→
   337→  const clearHistory = (): void => {
   338→    if (history.length === 0) return;
   339→    setHistory([]);
   340→    setConfirmClear(false);
   341→    toast({ title: "История очищена", duration: 3000 });
   342→  };
   343→

   345→
   346→  const copyId = async (value: string): Promise<void> => {

   348→      await navigator.clipboard.writeText(value);
   349→      toast({ title: "ID скопирован", description: value, variant: "success", duration: 2000 });
   350→    } catch {
   351→      toast({ title: "Не удалось скопировать ID", variant: "destructive", duration: 2500 });
   352→    }
   353→  };
   354→
   355→  const printSheet = printJob
   356→    ? createPortal(
   357→        printJob.batch ? (

   360→            <div className="mx-auto grid w-fit grid-cols-2 gap-x-[1cm] gap-y-[1cm] py-[0.5cm]">
   361→              {printJob.images.map((url, i) => (
   362→                <figure key={i} className="m-0 break-inside-avoid">
   363→                  <img
   364→                    src={url}
   365→                    alt={`Метка ${printJob.labels[i]} для печати`}
   366→                    style={{ width: `${printJob.sizeCm}cm`, height: "auto" }}
   367→                    onLoad={markImgLoaded}
   368→                    onError={markImgLoaded}
   369→                  />

   381→              onError={markImgLoaded}
   382→            />
   383→          </div>
   384→        ),
   385→        document.body
   386→      )
   387→    : null;
   388→
   389→  const downloadDisabled = !name.trim();
   390→
   391→  return (
   392→    <>
   393→      {printSheet}
   394→
   395→      <div className="grid items-start gap-6 lg:grid-cols-[minmax(320px,400px)_1fr]">
   396→        {/* Панель метки */}
   397→        <Card className="anim-rise rounded-[28px] border-stone-200/80 bg-white shadow-soft">

   399→            {/* Название — единственный ввод: задаёт и подпись, и уникальный узор */}

   421→                  value={name}
   422→                  maxLength={60}
