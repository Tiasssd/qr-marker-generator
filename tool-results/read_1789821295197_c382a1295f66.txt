    60→/** Задание на печать: одна метка или лист со всеми метками из истории */
    61→interface PrintJob {
    62→  /** data-URL картинок меток */
    63→  images: string[];
    64→  /** Понятные подписи (названия) — для alt */
    65→  labels: string[];
    66→  /** Физическая ширина одной метки на бумаге, см */

    72→function signature(it: MarkerConfig): string {
    73→  // ID и есть идентичность метки: «Молния» и «молния» — один и тот же узор
    74→  return it.id;
    75→}
    76→
    77→/* История: v2 (ID выводится из названия) с миграцией из v1 */
    78→function loadHistory(): HistoryItem[] {
    79→  try {
    80→    const raw =
    81→      window.localStorage.getItem(HISTORY_KEY) ??
    82→      window.localStorage.getItem("ar-marker-studio:history:v1");
    83→    if (!raw) return [];
    84→    const parsed: unknown = JSON.parse(raw);
    85→    if (!Array.isArray(parsed)) return [];
    86→    const items = parsed
    87→      .filter(
    88→        (it): it is HistoryItem =>
    89→          Boolean(it) &&
    90→          typeof it === "object" &&
    91→          typeof (it as HistoryItem).name === "string"
    92→      )
    93→      // Название — теперь идентичность метки: ID пересчитываем из него
    94→      .map((it) => {
    95→        const name = it.name.trim().replace(/\s+/g, " ");
    96→        return { name, id: deriveId(name), ts: typeof it.ts === "number" ? it.ts : Date.now() };
    97→      })
    98→      .filter((it) => it.name.length > 0);
    99→    return items.slice(0, 40);
   100→  } catch {
   101→    return [];
   102→  }
   103→}

   176→      dpr: Math.min(window.devicePixelRatio || 1, 2),

   315→      toast({ title: "Не удалось создать PNG", variant: "destructive", duration: 3500 });
   316→    }
   317→  };
   318→
   319→  const deleteItem = (item: HistoryItem): void => {

   324→      duration: 5000,

   333→    });
   334→  };

   336→  const clearHistory = (): void => {

   338→    setHistory([]);
   339→    setConfirmClear(false);
   340→    toast({ title: "История очищена", duration: 3000 });
   341→  };
   342→

   344→
   345→  const copyId = async (value: string): Promise<void> => {

   347→      await navigator.clipboard.writeText(value);
   348→      toast({ title: "ID скопирован", description: value, variant: "success", duration: 2000 });

   350→      toast({ title: "Не удалось скопировать ID", variant: "destructive", duration: 2500 });
   351→    }
   352→  };
   353→
   354→  const printSheet = printJob
   355→    ? createPortal(
   356→        printJob.batch ? (

   359→            <div className="mx-auto grid w-fit grid-cols-2 gap-x-[1cm] gap-y-[1cm] py-[0.5cm]">
   360→              {printJob.images.map((url, i) => (
   361→                <figure key={i} className="m-0 break-inside-avoid">
   362→                  <img
   363→                    src={url}
   364→                    alt={`Метка ${printJob.labels[i]} для печати`}
   365→                    style={{ width: `${printJob.sizeCm}cm`, height: "auto" }}

   383→        ),
   384→        document.body
   385→      )
   386→    : null;
   387→
   388→  const downloadDisabled = !name.trim();
   389→
   390→  return (

   393→
   394→      <div className="grid items-start gap-6 lg:grid-cols-[minmax(320px,400px)_1fr]">
   395→        {/* Панель метки */}
   396→        <Card className="anim-rise border-white/10 bg-neutral-900/60 backdrop-blur">

   420→                  value={name}
   421→                  maxLength={60}
   422→                  onChange={(e) => setName(e.target.value)}
   423→                  onKeyDown={(e) => {
   424→                    if (e.key === "Enter") void handleDownload();

   499→        {/* Предпросмотр */}
