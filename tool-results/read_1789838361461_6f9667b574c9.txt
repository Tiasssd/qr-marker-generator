    72→    72→function signature(it: MarkerConfig): string {
    73→    73→  // ID и есть идентичность метки: «Молния» и «молния» — один и тот же узор
    74→    74→  return it.id;
    75→    75→}

    77→    77→/* История: v2 (ID выводится из названия) с миграцией из v1 */
    78→    78→function loadHistory(): HistoryItem[] {
    79→    79→  try {
    80→    80→    const raw =
    81→    81→      window.localStorage.getItem(HISTORY_KEY) ??
    82→    82→      window.localStorage.getItem("ar-marker-studio:history:v1");
    83→    83→    if (!raw) return [];
    84→    84→    const parsed: unknown = JSON.parse(raw);
    85→    85→    if (!Array.isArray(parsed)) return [];
    86→    86→    const items = parsed
    87→    87→      .filter(
    88→    88→        (it): it is HistoryItem =>
    89→    89→          Boolean(it) &&
    90→    90→          typeof it === "object" &&
    91→    91→          typeof (it as HistoryItem).name === "string"
    92→    92→      )
    93→    93→      // Название — теперь идентичность метки: ID пересчитываем из него
    94→    94→      .map((it) => {
    95→    95→        const name = it.name.trim().replace(/\s+/g, " ");
    96→    96→        return { name, id: deriveId(name), ts: typeof it.ts === "number" ? it.ts : Date.now() };
    97→    97→      })
    98→    98→      .filter((it) => it.name.length > 0);
    99→    99→    return items.slice(0, 40);
   100→   100→  } catch {
   101→   101→    return [];
   102→   102→  }
   103→   103→}
