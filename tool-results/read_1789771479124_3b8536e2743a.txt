    91→    91→function signature(it: MarkerConfig): string {
    92→    92→  return `${it.id}|${it.name}|${it.style}|${it.frame}|${it.showId}`;
    93→    93→}

    95→    95→/* Загрузка истории из localStorage — только на клиенте (студия без SSR) */
    96→    96→function loadHistory(): HistoryItem[] {
    97→    97→  try {
    98→    98→    const raw = window.localStorage.getItem(HISTORY_KEY);
    99→    99→    if (!raw) return [];
   100→   100→    const parsed: unknown = JSON.parse(raw);
   101→   101→    if (!Array.isArray(parsed)) return [];
   102→   102→    const items = parsed.filter(
   103→   103→      (it): it is HistoryItem =>
   104→   104→        Boolean(it) &&
   105→   105→        typeof it === "object" &&
   106→   106→        typeof (it as HistoryItem).id === "string" &&
   107→   107→        (it as HistoryItem).id.length > 0 &&
   108→   108→        typeof (it as HistoryItem).name === "string" &&
   109→   109→        typeof (it as HistoryItem).ts === "number" &&
   110→   110→        typeof (it as HistoryItem).frame === "boolean" &&
   111→   111→        typeof (it as HistoryItem).showId === "boolean" &&
   112→   112→        ["classic", "rounded", "dots"].includes((it as HistoryItem).style)
   113→   113→    );
   114→   114→    return items.slice(0, 40);
   115→   115→  } catch {
   116→   116→    return [];
   117→   117→  }
   118→   118→}
